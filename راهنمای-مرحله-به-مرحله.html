CREATE TRIGGER trg_lines_no_edit_posted
BEFORE UPDATE ON journal_lines
WHEN EXISTS (SELECT 1 FROM journal_entries e
             WHERE e.id = OLD.entry_id AND e.status = 'posted')
BEGIN
    SELECT RAISE(ABORT, 'سند قطعی‌شده قابل ویرایش نیست؛ از سند اصلاحی استفاده کنید');
END;

-- ۲) سند فقط در صورت تراز بودن قطعی می‌شود
CREATE TRIGGER trg_entry_must_balance
BEFORE UPDATE OF status ON journal_entries
WHEN NEW.status = 'posted' AND OLD.status <> 'posted'
BEGIN
    SELECT RAISE(ABORT, 'سند تراز نیست: جمع بدهکار و بستانکار برابر نیست')
    WHERE (SELECT COALESCE(SUM(debit),0)  FROM journal_lines WHERE entry_id = NEW.id)
       <> (SELECT COALESCE(SUM(credit),0) FROM journal_lines WHERE entry_id = NEW.id)
       OR (SELECT COUNT(*) FROM journal_lines WHERE entry_id = NEW.id) < 2;
END;
محدودیتچرا
debit INTEGER CHECK (>= 0)مبالغ فقط عدد صحیح — استفاده از float در پول ممنوع
CHECK (debit = 0 OR credit = 0)یک سطر نمی‌تواند همزمان بدهکار و بستانکار باشد
CHECK (debit > 0 OR credit > 0)سطر بی‌اثر مجاز نیست

# تراکنش اتمیِ تودرتو (SQLite تراکنش تو‌در‌تو ندارد ← از SAVEPOINT استفاده شد)
@contextmanager
def tx(con):
    con.execute(f"SAVEPOINT sp_{n}")
    try:
        yield con
        con.execute("RELEASE sp_x")
    except Exception:
        con.execute("ROLLBACK TO sp_x"); con.execute("RELEASE sp_x"); raise
نکتهٔ مهم: هر عملیات مالی (فاکتور، دریافت، پرداخت) چندین جدول را همزمان تغییر می‌دهد (سند + سطرهای سند + حرکت انبار + به‌روزرسانی فاکتور). بدون تراکنش اتمی، یک خطای میانی دفاتر را نیمه‌ثبت و نامتوازن می‌کند. همهٔ این عملیات در with db.tx(con): انجام می‌شوند.

۴.۳ قیمت‌گذاری موجودی: میانگین موزون

def stock_state(con, item_id, upto_date=None):
    """بازسازی موجودی و بهای تمام‌شده با پخشِ تمام حرکت‌ها"""
    qty, avg = 0.0, 0.0
    for m in con.execute("... ORDER BY move_date, id"):
        q = float(m["qty"])
        if q > 0:                                    # ورود به انبار
            total = qty * avg + q * float(m["unit_cost"])
            qty  += q
            avg   = total / qty if qty else 0.0
        else:                                        # خروج از انبار
            qty -= -q
    return round(qty, 4), int(round(avg))

گزارش جدیدی که مطالبات/بدهی‌های باز را در بازه‌های جاری، ۱ تا ۳۰، ۳۱ تا ۶۰، ۶۱ تا ۹۰ و بیش از ۹۰ روز تفکیک می‌کند — ملاک، تاریخ سررسید فاکتور است و در صورت نبود، تاریخ فاکتور به‌علاوهٔ مهلتِ پیش‌فرض (۳۰ روز، قابل تنظیم).

۹.۴ قفل دوره مالی

# پیش از ثبت هر سند، قفل بودنِ ماهِ شمسی بررسی می‌شود
def _check_period_lock(con, fy, jdate):
    month = int(str(jdate).split("/")[1])
    if period_locked(con, fy["id"], month):
        raise AccountingError(f"ماه {month} از «{fy['title']}» قفل شده است")

قفل و باز کردنِ ماه‌ها از برگهٔ تنظیمات انجام می‌شود (ویژهٔ مدیر سیستم).

۱۰ رفعِ خطایِ ساختِ exe در ویندوز به‌روزشده

فایل build_windows.bat روی برخی سیستم‌ها با خطا متوقف می‌شد. چهار علتِ ریشه‌ای داشت که همگی برطرف شدند:

علتتوضیح و رفع
سرِخط‌های LF به‌جای CRLF cmd.exe فایل‌های بچ را فقط با CRLF درست می‌خواند؛ با LF ممکن است دستورها نیمه‌کاره یا خراب اجرا شوند. فایل اکنون با CRLF ذخیره شده است.
متنِ فارسیِ UTF-8 در فایل بچ cmd متنِ فارسی را بر اساس کدپیج (مثلاً ۱۲۵۶) تعبیر می‌کند و نویسه‌ها را خراب می‌کند؛ این خرابی می‌تواند خودِ دستور را هم مخدوش کند. پیام‌های فایل اکنون انگلیسی (ASCII خالص) هستند.
نبودِ cd /d "%~dp0" اگر فایل از جای دیگری اجرا شود (مثلاً Run as administrator که از System32 شروع می‌شود)، main.py پیدا نمی‌شود: ERROR: Script file 'main.py' does not exist. فایل ابتدا به پوشهٔ خودش می‌رود.
pyinstaller در PATH نیست فراخوانیِ مستقیمِ pyinstaller معمولاً با not recognized شکست می‌خورد؛ باید python -m PyInstaller بود.

۱۰.۱ روشِ جدید

کارِ اصلی به یک اسکریپتِ پایتونی منتقل شد تا به هیچ محدودیتی از cmd وابسته نباشد. build_windows.bat اکنون فقط وابستگی‌ها را نصب می‌کند و آن را صدا می‌زند.

:: روشِ پیشنهادی (دابل‌کلیک هم می‌توان کرد)
python build_exe.py

:: گزینه‌ها
python build_exe.py --onefile     :: یک فایل exe یکپارچه
python build_exe.py --console     :: همراه کنسول برای دیدنِ خطاها
python build_exe.py --dry-run     :: فقط نمایشِ فرمان

این اسکریپت پیش از ساخت، فایل schema.sql را درون یک ماژولِ پایتونی (schema_embedded.py) جاسازی می‌کند؛ بنابراین حتی اگر این فایل در بستهٔ exe گم شود، برنامه می‌تواند پایگاه داده را بسازد (یکی از خطاهای رایجِ اجرای exe همین بود).

۱۰.۳ اعتبارسنجی انجام‌شده

دومین باگ واقعی که هنگامِ اعتبارسنجی کشف و برطرف شد: نامِ فایلِ پشتیبان فقط تا دقتِ «ثانیه» بود (backup-%Y%m%d-%H%M%S.db)؛ بنابراین اگر دو پشتیبان در یک ثانیه گرفته می‌شد، دومی اولی را بی‌سر و صدا بازنویسی می‌کرد یعنی نسخهٔ قبلی عملاً از دست می‌رفت. این اشکال با یکتا کردنِ نامِ فایل برطرف شد (۱۰ پشتیبانِ پیاپی اکنون ۱۰ فایلِ مجزا می‌سازد) و با تست پوشش داده شد.

همچنین اجرای تست‌ها پیش از این نسخه‌های پشتیبان را کنارِ دادهٔ واقعیِ برنامه می‌نوشت؛ اکنون با متغیرِ ACC_DATA_DIR کاملاً ایزوله است.
